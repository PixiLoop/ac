'use client'

import { useEffect, useState, useRef } from 'react'
import { useParams } from 'next/navigation'
import { api } from '../../../../../lib/api-client'
import { AdminShell } from '../../../../../components/layout/AdminShell'

export default function CardEditPage() {
  const params = useParams()

  const [card, setCard] = useState<any>(null)
  const [users, setUsers] = useState<any[]>([])

  const [title, setTitle] = useState('')
  const [description, setDescription] = useState('')
  const [comment, setComment] = useState('')

  const [addingItemChecklistId, setAddingItemChecklistId] = useState<string | null>(null)
  const [newItemText, setNewItemText] = useState('')

  const [activePanel, setActivePanel] = useState<
    null | 'members' | 'checklist' | 'dates' | 'attachment' | 'labels'
  >(null)
  const titleRef = useRef<HTMLInputElement | null>(null)
  const panelRef = useRef<HTMLDivElement | null>(null)

  const [popupLabelName, setPopupLabelName] = useState('')
  const [popupLabelColor, setPopupLabelColor] = useState('#22c55e')

  const [newChecklistTitle, setNewChecklistTitle] = useState('')
  const [newLabelName, setNewLabelName] = useState('')
  const [newLabelColor, setNewLabelColor] = useState('#22c55e')
  const [dueDate, setDueDate] = useState('')
  const [search, setSearch] = useState('')
  const [editingCommentId, setEditingCommentId] = useState<string | null>(null)
  const [editingCommentText, setEditingCommentText] = useState('')

  const toLocalInput = (utcString: string): string => {
    const d = new Date(utcString)
    const pad = (n: number) => String(n).padStart(2, '0')
    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
  }

  const loadCard = async () => {
    const res = await api.get(`/job-boards/cards/${params.id}`)
    const data = res.data?.data ?? res.data
    setCard(data)
    setTitle(data.title)
    setDescription(data.description || '')
    setDueDate(data.dueDate ? toLocalInput(data.dueDate) : '')
  }

  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (
        panelRef.current &&
        !panelRef.current.contains(event.target as Node)
      ) {
        setActivePanel(null)
      }
    }

    document.addEventListener('mousedown', handleClickOutside)
    return () => {
      document.removeEventListener('mousedown', handleClickOutside)
    }
  }, [])

  useEffect(() => {
    const load = async () => {
      const u = await api.get('/users')
      setUsers(u.data?.data?.items ?? [])
      await loadCard()
    }
    load()
  }, [params.id])

  useEffect(() => {
    const handleClickOutsideTitle = async (event: MouseEvent) => {
      if (
        titleRef.current &&
        !titleRef.current.contains(event.target as Node)
      ) {
        if (card && title !== card.title) {
          await api.patch(`/job-boards/cards/${card.id}`, {
            title,
          })
          await loadCard()
        }
      }
    }

    document.addEventListener('mousedown', handleClickOutsideTitle)

    return () => {
      document.removeEventListener('mousedown', handleClickOutsideTitle)
    }
  }, [title, card])

  const saveCard = async () => {
    await api.patch(`/job-boards/cards/${card.id}`, {
      title,
      description,
      dueDate: dueDate || null,
    })
    await loadCard()
  }

  const toggleMember = async (userId: string, assigned: boolean) => {
    if (assigned) {
      await api.delete(`/job-boards/cards/${card.id}/members/${userId}`)
    } else {
      await api.post(`/job-boards/cards/${card.id}/members/${userId}`)
    }
    await loadCard()
  }

  const addChecklist = async () => {
    if (!newChecklistTitle) return
    await api.post(`/job-boards/cards/${card.id}/checklists`, {
      title: newChecklistTitle
    })
    setNewChecklistTitle('')
    setActivePanel(null)
    await loadCard()
  }

  const toggleItem = async (item: any) => {
    await api.patch(`/job-boards/checklist-items/${item.id}`, {
      isCompleted: !item.isCompleted
    })
    await loadCard()
  }

  const addLabel = async () => {
    await api.post(`/job-boards/cards/${card.id}/labels`, {
      name: newLabelName,
      color: newLabelColor
    })
    setNewLabelName('')
    await loadCard()
  }

  const addComment = async () => {
    if (!comment) return
    await api.post(`/job-boards/cards/${card.id}/comments`, {
      message: comment
    })
    setComment('')
    await loadCard()
  }

  const formatDate = (date: string) => {
    const d = new Date(date)

    const now = new Date()
    const diff = now.getTime() - d.getTime()

    const minutes = Math.floor(diff / 60000)
    const hours = Math.floor(diff / 3600000)
    const days = Math.floor(diff / 86400000)

    if (minutes < 1) return 'Just now'
    if (minutes < 60) return `${minutes} min ago`
    if (hours < 24) return `${hours} hr ago`
    if (days < 7) return `${days} days ago`

    return d.toLocaleString()
  }

  if (!card) return null

  return (
    <AdminShell title="">

      <div className="max-w-7xl mx-auto flex gap-6 py-6 h-[calc(100vh-120px)] overflow-hidden bg-gray-50 rounded-xl px-6">
        {/* LEFT */}
        <div className="flex-1 space-y-3 overflow-y-auto pr-2">
          {/* HEADER */}
          <div className="bg-white p-4 rounded-lg shadow-sm relative">

            <div className="flex items-center gap-2 mb-3">
              <div className="w-3 h-3 rounded-full bg-gray-400" />
              <span className="text-sm text-gray-600 font-medium">
                {card.list?.name}
              </span>
            </div>

            <input
              ref={titleRef}
              value={title}
              onChange={e => setTitle(e.target.value)}
              className="w-full text-2xl font-semibold border-b outline-none pb-2"
            />

            {/* ACTION BAR */}
            <div className="flex gap-2 mt-4 flex-wrap">
              <button onClick={() =>
                setActivePanel(activePanel === 'members' ? null : 'members')
              } className="btn">Members</button>
              <button onClick={() =>
                setActivePanel(activePanel === 'checklist' ? null : 'checklist')
              } className="btn">Checklist</button>
              <button onClick={() =>
                setActivePanel(activePanel === 'dates' ? null : 'dates')
              } className="btn">Dates</button>
              <button onClick={() =>
                setActivePanel(activePanel === 'attachment' ? null : 'attachment')
              } className="btn">Attachment</button>

              <button onClick={() =>
                setActivePanel(activePanel === 'labels' ? null : 'labels')
              } className="btn">
                Labels
              </button>

            </div>

            {/* MEMBERS DROPDOWN */}
            {activePanel === 'members' && (
              <div
                ref={panelRef}
                className="absolute bg-white shadow-xl border rounded-xl p-4 mt-3 w-80 z-50"
              >

                <h4 className="font-semibold mb-3">Members</h4>

                <input
                  placeholder="Search members"
                  className="border rounded-lg p-2 w-full mb-3"
                  onChange={(e) => setSearch(e.target.value)}
                />

                <div className="max-h-56 overflow-y-auto space-y-2">
                  {users
                    .filter(u =>
                      u.displayName?.toLowerCase()
                        .includes((search || '').toLowerCase())
                    )
                    .map(u => {
                      const assigned = card.members?.some((m: any) => m.userId === u.id)

                      return (
                        <div
                          key={u.id}
                          onClick={() => toggleMember(u.id, assigned)}
                          className={`flex justify-between items-center p-2 rounded-lg cursor-pointer
                ${assigned ? 'bg-blue-100' : 'hover:bg-gray-100'}
              `}
                        >
                          <span>{u.displayName}</span>
                          {assigned && <span className="text-blue-600 text-sm">✓</span>}
                        </div>
                      )
                    })}
                </div>
              </div>
            )}

            {/* CHECKLIST DROPDOWN */}
            {activePanel === 'checklist' && (
              <div
                ref={panelRef}
                className="absolute bg-white shadow-xl border rounded-xl p-4 mt-3 w-80 z-50"
              >
                <input
                  placeholder="Checklist title"
                  className="border p-2 w-full mb-2"
                  value={newChecklistTitle}
                  onChange={e => setNewChecklistTitle(e.target.value)}
                />
                <button onClick={addChecklist} className="btn-primary w-full">Add</button>
              </div>
            )}

            {/* DATES DROPDOWN */}
            {activePanel === 'dates' && (
              <div
                ref={panelRef}
                className="absolute bg-white shadow-xl border rounded-xl p-5 mt-3 w-80 z-50"
              >

                <h4 className="font-semibold mb-3">Dates</h4>

                <div className="mb-3">
                  <label className="text-sm text-gray-500">Due date</label>
                  <input
                    type="datetime-local"
                    value={dueDate}
                    onChange={(e) => setDueDate(e.target.value)}
                    className="w-full border rounded-lg p-2 mt-1"
                  />
                </div>

                <div className="flex justify-between mt-4">
                  <button
                    onClick={async () => {
                      await api.patch(`/job-boards/cards/${card.id}`, {
                        dueDate: null
                      })
                      setDueDate('')
                      setActivePanel(null)
                      await loadCard()
                    }}
                    className="text-red-500 text-sm"
                  >
                    Remove
                  </button>

                  <button
                    onClick={async () => {
                      await api.patch(`/job-boards/cards/${card.id}`, {
                        dueDate: dueDate ? new Date(dueDate).toISOString() : null
                      })
                      setActivePanel(null)
                      await loadCard()
                    }}
                    className="bg-blue-600 text-white px-4 py-1 rounded-lg text-sm"
                  >
                    Save
                  </button>
                </div>
              </div>
            )}

            {activePanel === 'labels' && (
              <div
                ref={panelRef}
                className="absolute bg-white shadow-xl border rounded-xl p-5 mt-3 w-80 z-50 space-y-4"
              >
                <h4 className="font-semibold">Labels</h4>

                {/* EXISTING LABELS */}
                <div className="space-y-2 max-h-40 overflow-y-auto">
                  {card.labels?.map((l: any) => (
                    <div
                      key={l.id}
                      className="flex items-center gap-3 p-2 rounded hover:bg-gray-100"
                    >
                      <input
                        type="checkbox"
                        checked
                        onChange={async () => {
                          await api.delete(
                            `/job-boards/cards/${card.id}/labels/${l.id}`
                          )
                          await loadCard()
                        }}
                        className="w-4 h-4 accent-blue-600"
                      />

                      <div
                        className="flex-1 text-white px-2 py-1 rounded text-sm"
                        style={{ backgroundColor: l.color }}
                      >
                        {l.name}
                      </div>
                    </div>
                  ))}
                </div>

                {/* ADD NEW LABEL */}
                <div className="flex items-center gap-2">
                  <input
                    placeholder="Label name"
                    value={popupLabelName}
                    onChange={(e) => setPopupLabelName(e.target.value)}
                    className="flex-1 border rounded-lg px-3 py-2 text-sm"
                  />

                  <input
                    type="color"
                    value={popupLabelColor}
                    onChange={(e) => setPopupLabelColor(e.target.value)}
                    className="w-10 h-10 p-0 border rounded cursor-pointer"
                  />

                  <button
                    onClick={async () => {
                      if (!popupLabelName.trim()) return

                      await api.post(`/job-boards/cards/${card.id}/labels`, {
                        name: popupLabelName,
                        color: popupLabelColor,
                      })

                      setPopupLabelName('')
                      setPopupLabelColor('#22c55e')

                      await loadCard()
                    }}
                    className="bg-blue-600 hover:bg-blue-700 text-white px-4 h-10 rounded-lg text-sm"
                  >
                    +
                  </button>
                </div>
              </div>
            )}


            {activePanel === 'attachment' && (
              <div
                ref={panelRef}
                className="absolute bg-white shadow-xl border rounded-xl p-5 mt-3 w-80 z-50 space-y-4"
              >
                <h4 className="font-semibold">Attach File</h4>

                <input
                  type="file"
                  onChange={async (e: any) => {
                    const file = e.target.files[0]
                    if (!file) return

                    const formData = new FormData()
                    formData.append('file', file)

                    await api.post(
                      `/job-boards/cards/${card.id}/attachments`,
                      formData,
                      { headers: { 'Content-Type': 'multipart/form-data' } }
                    )

                    setActivePanel(null)
                    await loadCard()
                  }}
                  className="w-full"
                />

                <button
                  onClick={() => setActivePanel(null)}
                  className="text-sm text-gray-500"
                >
                  Cancel
                </button>
              </div>
            )}
          </div>

          {/* SELECTED MEMBERS */}

          {card.members?.length > 0 && (
            <div className="mt-2">
              <div className="text-sm text-gray-600 mb-2 font-medium">
                Members
              </div>

              <div className="flex gap-2 flex-wrap items-center">

                {card.members?.length === 0 && (
                  <span className="text-sm text-gray-400">
                    No members added
                  </span>
                )}


                {card.members?.map((m: any) => (
                  <div
                    key={m.id}
                    className="w-8 h-8 rounded-full bg-blue-500 text-white flex items-center justify-center text-xs font-semibold relative group"
                  >
                    <span> <span>
                      {m.user?.displayName
                        ?.split(' ')
                        .map((n: string) => n[0])
                        .join('')
                        .substring(0, 2)
                        .toUpperCase()}
                    </span></span>

                    <button
                      onClick={async () => {
                        await api.delete(
                          `/job-boards/cards/${card.id}/members/${m.userId}`
                        )
                        await loadCard()
                      }}
                      className="absolute -top-1 -right-1 bg-white text-gray-600 text-[10px] rounded-full w-4 h-4 hidden group-hover:flex items-center justify-center shadow"
                    >
                      ✕
                    </button>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* LABELS */}
          <div className="bg-white p-4 rounded-lg shadow-sm">


            <div className="flex justify-between items-center mb-3">
              <h4 className="font-semibold">Labels</h4>
              {/* <button
    onClick={() =>
      setActivePanel(activePanel === 'labels' ? null : 'labels')
    }
    className="text-sm text-blue-600"
  >
    + Add
  </button> */}
            </div>


            <div className="flex gap-2 flex-wrap mb-2 items-center">

              {card.labels?.length === 0 && (
                <span className="text-sm text-gray-400">
                  No labels added
                </span>
              )}
              {card.labels?.map((l: any) => (
                <div
                  key={l.id}
                  className="px-3 py-1 text-[11px] font-medium text-white rounded"
                  style={{ backgroundColor: l.color }}
                >
                  <span>{l.name}</span>

                  <button
                    onClick={async () => {
                      await api.delete(`/job-boards/cards/${card.id}/labels/${l.id}`)
                      await loadCard()
                    }}
                    className="text-white text-xs ml-1 hover:opacity-70"
                  >
                    ✕
                  </button>
                </div>
              ))}
            </div>
            <div className="flex gap-2">
              <input
                placeholder="Label name"
                value={newLabelName}
                onChange={e => setNewLabelName(e.target.value)}
                className="border p-2 text-sm"
              />
              <input
                type="color"
                value={newLabelColor}
                onChange={e => setNewLabelColor(e.target.value)}
              />
              <button onClick={addLabel} className="btn-primary">+</button>
            </div>
          </div>

          {/* DESCRIPTION */}
          <div className="bg-white p-4 rounded-lg shadow-sm">
            <textarea
              rows={6}
              value={description}
              onChange={e => setDescription(e.target.value)}
              className="w-full bg-gray-50 rounded-md px-3 py-2 text-sm resize-y min-h-[120px] focus:bg-white focus:ring-1 focus:ring-blue-500 leading-relaxed"
              placeholder="Add a more detailed description..."
            />
          </div>

          {/* CHECKLISTS */}

          <h2 className="text-lg font-semibold mt-4 mb-3">
            Checklist
          </h2>

          {card.checklists?.length === 0 && (
            <div className="text-sm text-gray-400 bg-white p-4 rounded-lg shadow-sm">
              No checklist added
            </div>
          )}

          {card.checklists?.map((cl: any) => {
            const total = cl.items.length
            const done = cl.items.filter((i: any) => i.isCompleted).length
            const percent = total === 0 ? 0 : Math.round((done / total) * 100)

            return (
              <div key={cl.id} className="bg-white p-4 rounded-lg shadow-sm">

                {/* HEADER */}
                <div className="flex justify-between items-center mb-2">
                  <h4 className="font-semibold">{cl.title}</h4>
                  <span className="text-sm text-gray-500">{percent}%</span>
                </div>

                {/* PROGRESS */}
                <div className="h-2 bg-gray-200 rounded mb-3">
                  <div
                    className="h-2 bg-blue-600 rounded transition-all"
                    style={{ width: `${percent}%` }}
                  />
                </div>

                {/* ITEMS */}
                <div className="space-y-3">
                  {cl.items.map((item: any) => (
                    <div
                      key={item.id}
                      className="flex items-center gap-3"
                    >
                      <input
                        type="checkbox"
                        checked={item.isCompleted}
                        onChange={() => toggleItem(item)}
                        className="w-4 h-4 accent-blue-600"
                      />

                      <span
                        className={`text-sm ${item.isCompleted
                          ? 'line-through text-gray-400'
                          : 'text-gray-700'
                          }`}
                      >
                        {item.content}
                      </span>
                    </div>
                  ))}
                </div>

                {/* ADD ITEM */}
                <div className="mt-3">

                  {addingItemChecklistId === cl.id ? (
                    <div className="flex gap-2">
                      <input
                        value={newItemText}
                        onChange={(e) => setNewItemText(e.target.value)}
                        placeholder="Enter item name"
                        className="border p-2 rounded w-full text-sm"
                        autoFocus
                      />

                      <button
                        onClick={async () => {
                          if (!newItemText.trim()) return

                          await api.post(
                            `/job-boards/checklists/${cl.id}/items`,
                            { content: newItemText }
                          )

                          setNewItemText('')
                          setAddingItemChecklistId(null)
                          await loadCard()
                        }}
                        className="bg-blue-600 text-white px-3 rounded text-sm"
                      >
                        Add
                      </button>

                      <button
                        onClick={() => {
                          setNewItemText('')
                          setAddingItemChecklistId(null)
                        }}
                        className="text-gray-500 text-sm"
                      >
                        Cancel
                      </button>
                    </div>
                  ) : (
                    <button
                      onClick={() => setAddingItemChecklistId(cl.id)}
                      className="bg-gray-100 hover:bg-gray-200 px-3 py-1 rounded text-sm"
                    >
                      Add an item
                    </button>
                  )}

                </div>

              </div>
            )
          })}

          <button onClick={saveCard} className="btn-primary">Save Changes</button>

        </div>

        {/* RIGHT SIDE COMMENTS */}
        <div className="w-80 bg-white border rounded-xl shadow-sm p-4 flex flex-col h-full">

          <h3 className="font-semibold mb-3">Comments and Activity</h3>

          <textarea
            placeholder="Write a comment..."
            value={comment}
            onChange={e => setComment(e.target.value)}
            className="w-full border p-2 mb-2"
          />

          <button onClick={addComment} className="btn-primary w-full mb-4">Comment</button>



          <div className="space-y-3 overflow-y-auto pr-2 flex-1 text-sm">
            {card.activities?.map((a: any) => (
              <div key={a.id} className="flex gap-3 items-start group">

                {/* Avatar */}
                <div className="w-8 h-8 rounded-full bg-gray-400 text-white flex items-center justify-center text-[11px] font-semibold shrink-0">
                  {a.user?.displayName
                    ?.split(' ')
                    .map((n: string) => n[0])
                    .join('')
                    .substring(0, 2)
                    .toUpperCase()}
                </div>

                {/* Content */}
                <div className="flex-1 leading-tight">

                  <div className="text-[13px]">
                    <span className="font-semibold">{a.user?.displayName}</span>{' '}
                    {a.type !== 'COMMENT' && (
                      <span className="text-gray-700">{a.message}</span>
                    )}
                  </div>

                  {a.type === 'COMMENT' && (
                    editingCommentId === a.id ? (
                      <div className="mt-1 space-y-1">
                        <textarea
                          className="w-full border rounded px-2 py-1 text-sm resize-none focus:ring-1 focus:ring-blue-500"
                          rows={3}
                          value={editingCommentText}
                          onChange={e => setEditingCommentText(e.target.value)}
                        />
                        <div className="flex gap-2">
                          <button
                            onClick={async () => {
                              await api.patch(`/job-boards/comments/${a.id}`, { message: editingCommentText })
                              setEditingCommentId(null)
                              await loadCard()
                            }}
                            className="bg-blue-600 text-white px-3 py-0.5 rounded text-xs"
                          >
                            Save
                          </button>
                          <button
                            onClick={() => setEditingCommentId(null)}
                            className="text-gray-500 text-xs"
                          >
                            Cancel
                          </button>
                        </div>
                      </div>
                    ) : (
                      <div className="mt-0.5 bg-gray-50 border rounded px-2 py-1.5 text-gray-800 text-[13px]">
                        {a.message}
                      </div>
                    )
                  )}

                  <div className="flex items-center gap-3 mt-0.5">
                    <span className="text-[11px] text-gray-400">{formatDate(a.createdAt)}</span>
                    {a.type === 'COMMENT' && editingCommentId !== a.id && (
                      <div className="hidden group-hover:flex items-center gap-2">
                        <button
                          onClick={() => { setEditingCommentId(a.id); setEditingCommentText(a.message) }}
                          className="text-[11px] text-gray-400 hover:text-blue-600"
                        >
                          Edit
                        </button>
                        <button
                          onClick={async () => {
                            if (confirm('Delete this comment?')) {
                              await api.delete(`/job-boards/comments/${a.id}`)
                              await loadCard()
                            }
                          }}
                          className="text-[11px] text-gray-400 hover:text-red-500"
                        >
                          Delete
                        </button>
                      </div>
                    )}
                  </div>

                </div>

              </div>
            ))}
          </div>


        </div>

      </div>

    </AdminShell>
  )
}
